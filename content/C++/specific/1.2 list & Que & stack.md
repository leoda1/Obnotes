## 2 Que
### 2.1 intrusive queue
在nccl内常见侵入式队列，这里以它的定义和使用方式来说明这是怎么，以及怎么用。
#### 定义
队列的结构只包含头尾指针，`T *T::*next` 是成员指针类型，指向类型 T 的 `next` 成员。
* 这就支持一个T内可以搞多个`next`成员
* 一个节点可以属于多个链表
所以说明在使用这个需要我们来决定输入节点的结构 T。
nccl还提供了多个基础增删改查；
```cpp
// 定义
template<typename T, T *T::*next>
//        ↑      ↑   ↑ ↑ ↑ ↑
//        │      │   │ │ │ └─ 成员名（编译时常量）
//        │      │   │ │ └─── 成员指针类型
//        │      │   │ └───── 指向 T 类型的指针
//        │      │   └─────── 成员指针的语法
//        │      └─────────── 类型参数
//        └────────────────── 模板关键字
struct ncclIntruQueue {
  T *head, *tail;
};


// 增删改查
template<typename T, T *T::*next>
inline void ncclIntruQueueConstruct(ncclIntruQueue<T,next> *me) {
  me->head = nullptr;
  me->tail = nullptr;
}

template<typename T, T *T::*next>
inline bool ncclIntruQueueEmpty(ncclIntruQueue<T,next> *me) {
  return me->head == nullptr;
}

template<typename T, T *T::*next>
inline T* ncclIntruQueueHead(ncclIntruQueue<T,next> *me) {
  return me->head;
}

template<typename T, T *T::*next>
inline T* ncclIntruQueueTail(ncclIntruQueue<T,next> *me) {
  return me->tail;
}

template<typename T, T *T::*next>
inline void ncclIntruQueueEnqueue(ncclIntruQueue<T,next> *me, T *x) {
  x->*next = nullptr;
  (me->head ? me->tail->*next : me->head) = x;
  me->tail = x;
}

template<typename T, T *T::*next>
inline void ncclIntruQueueEnqueueFront(ncclIntruQueue<T,next> *me, T *x) {
  if (me->head == nullptr) me->tail = x;
  x->*next = me->head;
  me->head = x;
}

template<typename T, T *T::*next>
inline bool ncclIntruQueueDelete(ncclIntruQueue<T,next> *me, T *x) {
  T *prev = nullptr;
  T *cur = me->head;
  bool found = false;

  while (cur) {
    if (cur == x) {
      found = true;
      break;
    }
    prev = cur;
    cur = cur->*next;
  }

  if (found) {
    if (prev == nullptr)
      me->head = cur->*next;
    else
      prev->*next = cur->*next;
    if (cur == me->tail)
      me->tail = prev;
  }
  return found;
}

template<typename T, T *T::*next>
inline T* ncclIntruQueueTryDequeue(ncclIntruQueue<T,next> *me) {
  T *ans = me->head;
  if (ans != nullptr) {
    me->head = ans->*next;
    if (me->head == nullptr) me->tail = nullptr;
  }
  return ans;
}

template<typename T, T *T::*next>
void ncclIntruQueueTransfer(ncclIntruQueue<T,next> *dst, ncclIntruQueue<T,next> *src) {
  (dst->tail ? dst->tail->next : dst->head) = src->head;
  if (src->tail) dst->tail = src->tail;
  src->head = nullptr;
  src->tail = nullptr;
}

template<typename T, T *T::*next>
inline T* ncclIntruQueueDequeue(ncclIntruQueue<T,next> *me) {
  T *ans = me->head;                                        // 步骤1
  me->head = ans->*next;                                    // 步骤2
  if (me->head == nullptr) me->tail = nullptr;              // 步骤3
  return ans;
}

```
#### 使用
例如调用上面的入队的函数：
```cpp
//参数说明：
// &comm->rmaCeInitTaskQueue是ncclIntruQueue<struct ncclRmaCeInitTask, &ncclRmaCeInitTask::next>*
// 会去匹配                    ncclIntruQueue<            T,             next>                 *me
// 得到 T = ncclRmaCeInitTask, next = &ncclRmaCeInitTask::next

// ceTask是ncclRmaCeInitTask*
ncclIntruQueueEnqueue(&comm->rmaCeInitTaskQueue, ceTask);
```
插入第一个节点 x1：
```cpp
步骤1: x1->next = nullptr
步骤2: me->head = x1  (因为 me->head 为空，所以执行 me->head = x1)
步骤3: me->tail = x1

结果:
me->head ──> [x1] -> nullptr
me->tail ──> [x1]
```
插入第二个节点 x2:
```cpp
步骤1: x2->next = nullptr
步骤2: me->tail->next = x2  (因为 me->head 不为空，所以执行 me->tail->next = x2)
步骤3: me->tail = x2

结果:
me->head ──> [x1] -> [x2] -> nullptr
me->tail ──> [x2]
```
插入第三个节点 x3：
```cpp
步骤1: x3->next = nullptr
步骤2: me->tail->next = x3
步骤3: me->tail = x3

结果:
me->head ──> [x1] -> [x2] -> [x3] -> nullptr
me->tail ──> [x3]
```
